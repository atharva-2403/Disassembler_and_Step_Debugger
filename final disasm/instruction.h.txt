//instruction.h - shared instruction struct: worklet 1 fills it, worklet 2 executes it, worklet 3 displays it
//no <stdlib.h> here: its register_t clashes with worklet 1's enum
#ifndef INSTRUCTION_H
#define INSTRUCTION_H

#include <stdint.h>
#include <stddef.h>
#include <stdbool.h>
#include "decoded_binary_framework.h"

//how the instruction changes eip
typedef enum {
    NORMAL,
    CONDITIONAL_BRANCH,
    JUMP,
    CALL,
    RET
} InstructionType;

//registers in x86 encoding order, same numbers as the ModR/M fields
typedef enum {
    REG_EAX = 0,
    REG_ECX = 1,
    REG_EDX = 2,
    REG_EBX = 3,
    REG_ESP = 4,
    REG_EBP = 5,
    REG_ESI = 6,
    REG_EDI = 7,
    REG_INVALID = 255    //no register
} RegName;

#define REG_COUNT 8

//memory operand: address = base + index*scale + disp
typedef struct {
    RegName base;       //REG_INVALID if none
    RegName index;      //REG_INVALID if none
    uint8_t scale;      //1, 2, 4 or 8
    int32_t disp;
} MemOperand;

typedef struct {
    uint32_t start_address;   //virtual address of the instruction
    size_t   length;          //size in bytes

    decoded_binary raw_decoded;  //raw bytes, for the listing

    char mnemonic[16];        //lowercase name, e.g. "mov"
    InstructionType type;
    uint32_t target_address;  //absolute jump/call target

    RegName dest_reg;         //first operand, REG_INVALID if none
    RegName src_reg;          //second operand, REG_INVALID if none
    int32_t immediate_val;
    bool    has_immediate;

    bool       is_memory_dest;   //first operand is memory
    MemOperand mem_dest;
    bool       is_memory_src;    //second operand is memory
    MemOperand mem_src;
} Instruction;

#endif
